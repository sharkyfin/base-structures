cmake_minimum_required(VERSION 3.24)
project(base_structures LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(BUILD_GUI "Build the browser AVL visualizer" ON)
include(CTest)

include(FetchContent)
FetchContent_Declare(pbplots
    URL https://github.com/InductiveComputerScience/pbPlots/archive/eb4cdc091c23c3143fc37791e0211022f0104a43.tar.gz
    URL_HASH SHA256=9a1e6a5dc43d372ed55145d6cc71926d2bbe3096e38b680136d2e0f1c09aeee7)
FetchContent_MakeAvailable(pbplots)
add_library(pbplots STATIC "${pbplots_SOURCE_DIR}/Cpp/pbPlots.cpp" "${pbplots_SOURCE_DIR}/Cpp/supportLib.cpp")
target_include_directories(pbplots SYSTEM PUBLIC "${pbplots_SOURCE_DIR}/Cpp")

add_library(structures_report STATIC benchmark/Report.cpp)
target_include_directories(structures_report PUBLIC .)
target_compile_options(structures_report PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(structures_report PRIVATE pbplots)
file(MAKE_DIRECTORY "${PROJECT_SOURCE_DIR}/benchmark/results")

if(BUILD_GUI)
    add_subdirectory(GUI)
endif()

add_executable(structures main.cpp)
target_include_directories(structures PRIVATE .)
target_compile_options(structures PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(structures PRIVATE structures_report)

add_executable(structures_benchmark benchmark/main.cpp)
target_include_directories(structures_benchmark PRIVATE .)
target_compile_options(structures_benchmark PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(structures_benchmark PRIVATE structures_report)

if(BUILD_TESTING)
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND)
        include(FetchContent)
        FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz)
        set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
        FetchContent_MakeAvailable(googletest)
    endif()
    add_executable(structures_tests tests.cpp GUI/TreeSession.cpp)
    target_include_directories(structures_tests PRIVATE .)
    target_compile_options(structures_tests PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(structures_tests PRIVATE GTest::gtest_main structures_report)
    include(GoogleTest)
    gtest_discover_tests(structures_tests)
endif()

if(ENABLE_SANITIZERS)
    foreach(target structures structures_benchmark structures_tests avl_visualizer structures_report pbplots)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endforeach()
endif()
