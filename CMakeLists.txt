cmake_minimum_required(VERSION 3.24)
project(base_structures LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(BUILD_GUI "Build the browser AVL visualizer" ON)
include(CTest)

if(BUILD_GUI)
    add_subdirectory(GUI)
endif()

add_executable(structures main.cpp)
target_include_directories(structures PRIVATE .)
target_compile_options(structures PRIVATE -Wall -Wextra -Wpedantic)

if(BUILD_TESTING)
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND)
        include(FetchContent)
        FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz)
        set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
        FetchContent_MakeAvailable(googletest)
    endif()
    add_executable(structures_tests tests.cpp GUI/TreeSession.cpp)
    target_include_directories(structures_tests PRIVATE .)
    target_compile_options(structures_tests PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(structures_tests PRIVATE GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(structures_tests)
endif()

if(ENABLE_SANITIZERS)
    foreach(target structures structures_tests avl_visualizer)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endforeach()
endif()
